=== рекурсия
function fact(n: int) returns (r: int)
{
  if (n == 0) r = 1; else r = n * fact(n - 1);
}
--- ожидается
fact(5) = (120)
fact(0) = (1)

=== деление и остаток как в JVM, деление на ноль
function d(a: int, b: int) returns (q: int, r: int)
{
  q = a / b;
  r = a % b;
}
--- ожидается
d(17, 5) = (3, 2)
d(-7, 2) = (-3, -1)
d(7, -2) = (-3, 1)
d(7, 0) = ошибка: деление на ноль

=== переполнение по модулю 2^32
function f(x: int) returns (r: int)
{
  r = x + 1;
}
--- ожидается
f(2147483647) = (-2147483648)
f(1) = (2)

=== сумма массива и выход за границы
function sum(a: int[], n: int) returns (s: int)
{
  var i = 0;
  s = 0;
  while (i < n) {
    s = s + a[i];
    i = i + 1;
  }
}
--- ожидается
sum([1, 2, 3], 3) = (6)
sum([1, 2, 3], 0) = (0)
sum([1, 2, 3], 4) = ошибка: выход за границы массива

=== короткая схема && защищает от выхода за границы
function find(a: int[], v: int) returns (k: int)
{
  k = 0;
  while (k < length(a) && a[k] != v) {
    k = k + 1;
  }
  if (k == length(a)) k = -1;
}
--- ожидается
find([4, 5, 6], 6) = (2)
find([4, 5, 6], 7) = (-1)
find([], 1) = (-1)

=== массивы копируются при обновлении
function f(a: int[]) returns (old: int[], new: int[])
{
  old = a;
  new = a;
  new[0] = 9;
}
--- ожидается
f([1, 2]) = ([1, 2], [9, 2])
f([]) = ошибка: выход за границы массива

=== нарушение assert
function f(x: int) returns (r: int)
{
  assert x > 0;
  r = x;
}
--- ожидается
f(5) = (5)
f(-1) = ошибка: нарушение assert

=== оптимизации внутри цикла
function f(n: int) returns (r: int)
{
  var i = 0;
  r = 0;
  while (i < n) {
    var k = 2 * 3;
    var u = i + 0;
    var w = u * k;
    r = r + w + u * k;
    i = i + 1;
  }
}
--- ожидается
f(4) = (72)
f(0) = (0)
