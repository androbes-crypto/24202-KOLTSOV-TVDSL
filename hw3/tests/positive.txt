=== вызовы функций, рекурсия и вызов функции, объявленной ниже
function fact(n: int) returns (r: int)
  requires n >= 0
{
  if (n == 0) r = 1; else r = n * fact(n - 1);
}

function main() returns (r: int)
{
  r = fact(5) + twice(3);
}

function twice(x: int) returns (y: int)

  requires b > 0
  ensures a == q * b + r

function abs(x: int) returns (y: int)
  requires x > -1000 && x < 1000
  ensures y >= 0 && (y == x || y == -x)
{
  if (x < 0) y = -x; else y = x;
}

=== условия циклов и проверок
function loop(n: int) returns (k: int)
  requires n >= 0
  ensures k == n
{
  k = 0;
  while (k < n)
    invariant 0 <= k && k <= n
    invariant !(k > n)
  {
    assume k >= 0;
    k = k + 1;
  }
  assert k == n;
}
--- ожидается
